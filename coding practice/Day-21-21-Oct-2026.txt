# 1. Ansible- Install nginx and enable service

---
- name: Install nginx and start service
  hosts: all
  become: yes

  tasks:
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: start service
      yum:
        name: nginx
        state: started
        enabled: yes

# Multi-Environment Pipeline

# Jenkins pipeline - deploy to Multiple environments

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
    }
    Stages {
        stage('Deploy to Dev') {
            steps {
                sh 'echo "Deploying $INSTITUTE_NAME to Dev environment"'
            }
        }
        stage('Deploy to Staging') {
            steps {
                sh 'echo "Deploying $INSTITUE_NAME to Staging environment"'
            }
        }
        stage('Deploy to Prod') {
            steps {
                sh 'echo "Deploying $INSTITUTE_NAME to Prod environment"'
            }
        }
    }
}

## GitLab CI- Multi-environment
You will learn how to **define multiple environment deployments in GitLab CI **.

stages:
  - deploy
 
variables:
  INSTITUTE_NAME : "Pathnex"

deploy-dev:
  stage: deploy
  image: alpine:latest
  script:
    - echo "Deploying $INSTITUTE_NAME to Dev envrironment"

deploy-staging:
  stage: deploy
  image: alpine:latest
  script:
    - echo "Deploying $INSTITUTE_NAME to staging environment"
  when: manual

deploy-prod:
  stage: deploy
  image: alpine:latest
  script:
    - echo "Deploying $INSTITUTE_NAME to prod environmment"
  when: manual
