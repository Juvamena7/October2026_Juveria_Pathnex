# VPC creation
resource "aws_vpc" "main" {
  cidr_block = "10.0.0.0/16"
  
  tags {
    Name = "Pathnex-vpc"
  }    
}

# Public-subnet

resource "aws_subnet" "public" {
  vpc_id                     = aws_vpc.main.id
  cidr_block                 = "10.0.1.0/24"
  map_public_ip_on_launch    = true

  tags {
    Name = "Pathnex_public_subnet"
  }

}

# Internet gateway
resource "aws_internet_gateway" "igw" {
  vpc_id    = aws_vpc.main.id

  tags = {
    Name = "Pathnex-igw"
  }
}

# Route-table
resource "aws_route_table" "public" {
  vpc_id  = aws_vpc.main.id

  route {
    cidr_block  = "0.0.0.0/0"
    gateway_id  = aws_internet_gateway.igw.id
  }
  tags {
    Name = "Pathnex-Public-RT"
  }
}
# subnet association
resource "aws_route_table_association" "public" {
  subnet_id       = aws_subnet.public.id
  route_table_id  = aws_route_table.public.id
}

## 2️⃣ Ansible — Install Nginx & Deploy HTML File

---
- name: Install Nginx & Deploy HTML file
  hosts: all
  become: yes

  tasks: 
    - name: Install Nginx
      yum: 
        name: nginx
        state: present
    - name: Deploy HTML file
      copy:
        src: index.html
        dest: /usr/share/nginx/html/index.html     

## 3️⃣ Kubernetes — Create Deployment + Service

apiVersion : apps/v1
kind: Deployment
metadata:
  name: pathnex-Deployment
spec:
  replicas: 2
  selector:
    matchLables:
      app: pathnex
  template:
    metadata:
      labels:
        app: pathnex  
    spec:
      containers:
        - name: nginx
          image: nginx:latest
          ports:
            - containerPort: 80
---
apiVersion: v1
kind: service
metadata:
  name: pathnex-service
spec:
  type: NodePort
  selector:
    app: pathnex
  ports:
    - port: 80
      targetPort: 80
      nodePort: 30080

# Scheduled Pipelines

## Jenkins Pipeline - Scheduled Build
You will learn how to **schedule a Jenkins Pipeline using cron **.

pipeline {
    agent any
    environment {
        INSTITUTE_NAME  = "Pathnex"
    }
    triggers {
        cron('H 2 * * *') // every day at 2 AM
    }
    stages {
        stage('Checkout') {
            steps {
                git url: 'https://github.com/Pathnex/sample-java-app.git'
            }
        }
        stage('Build') {
            steps {
                sh 'mvn clean package'
            }
        }
    }
}
