- name: Install nginx and start service
  hosts: all
  become: yes

  tasks: 
    - name: Install nginx
      yum:
        name: nginx
        state: present
    - name: Start and enable nginx service
      service:
        name: nginx
        state: started
        enabled: yes
    - name: Deploy Welcome HTML page
      copy:
        content: "Welcome to Pathnex"
        dest: /usr/share/nginx/html/index.html
        owner: root
        group: root
        mode: 0644


## kubernetes- Create Deploymnet with PVC and expose port via NodePort

apiVersion: apps/v1
kind: Deployment
metadata:
  name: Pathnex-Deployment
spec:
  replicas: 3
  selectors:
    matchLabels: 
      app: pathnex
  template:
    metadata:
      labels:
        app: pathnex
    spec:
      containers:
        - name: nginx
          image: nginx-latest
          ports:
            - containerPort: 80
      volumes:
        - name: pathnex-volume
          persistentVolumeClaim:
            claimName: pathnex-pvc
---
# NodePort service 
apiVersion: v1
kind: Service
metadata: 
  name: pathnex-nodeport
spec:
  type: NodePort
  selector:
    app: pathnex
  ports:
    - port: 80
      targetPort: 80
      nodePort: 30080
---
# PersistentVolumeClaim (PVC)
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: pathnex-pvc
spec:
  accessModes:
    - ReadWritOnce
  resources:
    requests:
      storage: 1Gi

## Shell Script

#!/bin/bash

echo "5-System checks"

echo "CPU utilisation of various process is: "
top | grep 'Cpu(s)' | awk '{print $2}'

echo "Memory utilisation of various process is: "
free -m | awk '/Mem/ {print $3 "/" $2 }'

echo "Disk usage is: "
df -h / | awk 'NR==2 {print $5, $4}'

echo "Ip address is: "
hostname -I | awk '{print $1}'

echo "Top processes running on system are: "

ps -eo pid,comm,%cpu --sort=-%cpu | sed -n '1,2p'


# Jenkins pipeline- All tasks combined for project work through jenkins pipeline

pipeline {
    agent any
    environment {
        INSTITUTE_NAME = "Pathnex"
        TEAM = "DevOps"
        ENV = "prod"
        PROJECT = "Pathnex-training"
    }
    stages {
        stage('Infrastructure') {
            steps {
                echo "Creating infrastructure through terraform for ${PROJECT}"
                sh 'terraform init'
                sh 'terraform apply'
            }
        }
        stage('Deployment') {
            steps {
                echo "Deploying application for ${INSTITUTE_NAME}"
                sh 'ansible-playbook -i inventoryfile install_ngix.yml'
                sh 'kubectl apply -f pathnex-app.yaml
            }
        }
        stage('Monitoring Checks') {
            steps {
                echo "Running system checks for ${TEAM}"
                sh "bash system_checks.sh"
            }
        }
    }
    post {
        success {
            echo "${PROJECT} deployed successfully in ${ENV}"
        }
        failure {
            echo "pipeline failed. Check the console logs."
        }
        always {
            echo "Pipeline execution completed  for ${INSTITUTE_NAME}"
        }
    }
}
